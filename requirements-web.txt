# Minimal runtime dependencies for serving the Flask dashboard (app/).
# The full requirements.txt is for the data pipeline / model training and
# includes heavy packages (Prophet, SHAP, Optuna, LightGBM,
# Selenium) this web process never imports -- installing all of that on a
# free-tier host risks hitting build time/RAM/disk limits for no reason.
# Pinned to 3.1.0 because NepseUnofficialApi (live NEPSE data) requires
# flask==3.1.0 exactly; leaving this at 3.0.0 makes pip fail to resolve.
flask==3.1.0
flask-cors==4.0.0
gunicorn==21.2.0

# User accounts / watchlist / portfolio (app/backend/accounts.py). DATABASE_URL points at
# a free hosted Postgres (Neon/Supabase); psycopg2-binary is its driver.
Flask-SQLAlchemy==3.1.1
psycopg2-binary==2.9.9
# Google sign-in (only active when GOOGLE_CLIENT_ID/SECRET are set)
Authlib==1.3.2
requests>=2.31
numpy==1.26.2
scikit-learn==1.3.2
xgboost==2.0.2
# No PyTorch here on purpose. Loading it and a model per request (the watchlist asks for many
# signals at once) used ~500 MB and crashed the 512 MB free plan. The signals and explanations
# are computed by the daily GitHub workflow (scripts/build_explanations.py) and the site only
# reads models/explanations.json. To score live on your own machine, `pip install torch` and
# leave LIVE_MODEL_SCORING unset (it is off on Render by default).

# Live NEPSE market data for /api/live. If a host can't install from git, or
# NEPSE blocks it, the dashboard degrades to end-of-day data automatically --
# see ml/scraper/live_market.py.
nepse @ git+https://github.com/basic-bgnr/NepseUnofficialApi
